```java
private static final String DELETE_STATUS_FIELD = "deleteStatus";
private static final String DELETE_TIMESTAMP_FIELD = "deleteTimestamp";

/**
 * 删除知识文档 - 两阶段删除确保一致性
 * 阶段1: 标记为删除中 -> 阶段2: 删除向量 -> 阶段3: 删除MongoDB记录
 */
public void deleteDocument(String documentId) {
    Query query = new Query(Criteria.where("documentId").is(documentId));
    KnowledgeDocument document = mongoTemplate.findOne(query, KnowledgeDocument.class);

    if (document == null) {
        log.warn("文档不存在, id: {}", documentId);
        return;
    }

    try {
        // 阶段1: 标记为删除中
        Update markDelete = new Update()
            .set(DELETE_STATUS_FIELD, "DELETING")
            .set(DELETE_TIMESTAMP_FIELD, LocalDateTime.now());
        mongoTemplate.updateFirst(query, markDelete, KnowledgeDocument.class);

        // 阶段2: 删除向量（重试3次）
        if (document.getVectorIds() != null && !document.getVectorIds().isEmpty()) {
            deleteVectorsWithRetry(document.getVectorIds(), 3);
        }

        // 阶段3: 删除MongoDB记录
        mongoTemplate.remove(query, KnowledgeDocument.class);
        log.info("知识文档已删除, id: {}", documentId);

    } catch (Exception e) {
        log.error("删除文档失败，已标记为DELETING状态，需要手动清理, id: {}, error: {}", documentId, e.getMessage());
        // 不再抛异常，防止事务回滚导致的状态不一致
        // 触发告警通知运维处理
    }
}

/**
 * 删除向量 - 带重试机制
 */
private void deleteVectorsWithRetry(List<String> vectorIds, int maxRetries) {
    int retries = 0;
    List<String> failedIds = new ArrayList<>();

    for (String vectorId : vectorIds) {
        try {
            embeddingStore.remove(vectorId);
        } catch (Exception e) {
            failedIds.add(vectorId);
            log.warn("删除向量失败, vectorId: {}, error: {}", vectorId, e.getMessage());

            // 重试
            if (retries < maxRetries) {
                retries++;
                try {
                    Thread.sleep(100 * retries); // 指数退避
                    embeddingStore.remove(vectorId);
                    failedIds.remove(vectorId);
                } catch (Exception retryErr) {
                    log.error("向量删除重试仍失败, vectorId: {}", vectorId);
                }
            }
        }
    }

    if (!failedIds.isEmpty()) {
        log.error("向量删除部分失败，孤立向量ID列表: {}", failedIds);
        // TODO: 发送告警到告警系统
    }
}
```

### B007: 预测 Tool 接入 daily_stock_analysis

**文件**: `src/main/java/com/ljl/ai/agent/tools/TimeSeriesPredictionTool.java`、`src/main/resources/application.yml`

当前项目原先调用未配置的 `POST /predict`。`daily_stock_analysis` 实际提供的是同步股票分析接口：

```http
POST http://localhost:8000/api/v1/analysis/analyze
Content-Type: application/json

{"stock_code":"600519","report_type":"detailed","async_mode":false,"notify":false}
```

Java Tool 现在调用该接口，并提取 `report.summary.trend_prediction`，同时返回当前价格、涨跌幅、操作建议和风险提示。原有 `horizon`、`model` 参数保留用于 Agent Tool 兼容，但该项目的预测不是独立 LSTM/Transformer 接口，因此结果模型标记为 `daily_stock_analysis`。

启动目标 Python 项目后配置：

```bash
PREDICTION_BASE_URL=http://localhost:8000
PREDICTION_TIMEOUT_SECONDS=180
```

前端请求中 `enableTools=true` 时，Agent 会根据问题调用 `predictStockTrend`；未设置地址时不会伪造预测，只返回未配置提示。

```java
/**
 * 处理文档并存储向量 - 带容错和补偿机制
 */
private List<String> processAndStoreDocument(KnowledgeDocument document) {
    Document doc = Document.from(document.getRawContent(), Metadata.from(Map.of(
        "documentId", document.getDocumentId(),
        "title", document.getTitle(),
        "documentType", document.getDocumentType() != null ? document.getDocumentType() : "",
        "source", document.getSource() != null ? document.getSource() : ""
    )));

    DocumentSplitter splitter = DocumentSplitters.recursive(
        knowledgeConfig.getChunk().getSize(),
        knowledgeConfig.getChunk().getOverlap()
    );
    List<TextSegment> segments = splitter.split(doc);
    log.info("文档分块完成, 共 {} 个片段", segments.size());

    // 处理向量 - 记录成功和失败的vectorIds
    List<String> successVectorIds = new ArrayList<>();
    List<Integer> failedSegmentIndices = new ArrayList<>();

    for (int i = 0; i < segments.size(); i++) {
        TextSegment segment = segments.get(i);
        try {
            Embedding embedding = embeddingModel.embed(segment).content();
            String vectorId = embeddingStore.add(embedding, segment);
            successVectorIds.add(vectorId);
        } catch (Exception e) {
            failedSegmentIndices.add(i);
            log.error("向量存储失败, segment索引: {}, error: {}", i, e.getMessage());
        }
    }

    // 检查是否有失败
    if (!failedSegmentIndices.isEmpty()) {
        log.warn("部分segment处理失败: {}/{}", failedSegmentIndices.size(), segments.size());

        // 如果成功率低于80%，补偿删除已添加的向量
        if (successVectorIds.size() < segments.size() * 0.8) {
            log.error("成功率过低，执行补偿操作删除已添加的向量");
            successVectorIds.forEach(vectorId -> {
                try {
                    embeddingStore.remove(vectorId);
                } catch (Exception e) {
                    log.error("补偿删除向量失败, vectorId: {}", vectorId);
                }
            });
            throw new RuntimeException(
                String.format("向量存储失败，成功率仅%.1f%%，已回滚",
                    (double) successVectorIds.size() / segments.size() * 100)
            );
        }

        // 成功率足够，返回成功的vectorIds
        log.info("向量存储完成, 成功: {}, 失败: {}", successVectorIds.size(), failedSegmentIndices.size());
    }

    return successVectorIds;
}
```

```java
/**
 * 同步飞书文档 - 使用乐观锁处理并发
 */
public KnowledgeDocument syncFeishuDocument(String docToken, String documentType, List<String> tags) {
    log.info("开始同步飞书文档, docToken: {}", docToken);

    String content = feishuClient.getDocumentContent(docToken);
    if (content == null || content.isEmpty()) {
        log.error("获取飞书文档内容失败, docToken: {}", docToken);
        return null;
    }

    var docMeta = feishuClient.getDocumentMeta(docToken);
    String title = docMeta != null ? docMeta.getString("title") : "未命名文档";

    // 使用原子操作确保并发安全
    int maxRetries = 3;
    int retry = 0;

    while (retry < maxRetries) {
        try {
            KnowledgeDocument existingDoc = findByFeishuDocToken(docToken);
            KnowledgeDocument document = existingDoc != null ? existingDoc : new KnowledgeDocument();

            // 设置文档属性
            document.setFeishuDocToken(docToken);
            document.setTitle(title);
            document.setDocumentType(documentType);
            document.setRawContent(content);
            document.setSource("FEISHU");
            document.setUrl("https://xxx.feishu.cn/docx/" + docToken);
            document.setTags(tags);
            document.setSyncTime(LocalDateTime.now());
            document.setEnabled(true);

            if (existingDoc == null) {
                document.setDocumentId(UUID.randomUUID().toString());
                document.setCreateTime(LocalDateTime.now());
                // @Version初始为null, MongoDB会设为0
            }
            document.setUpdateTime(LocalDateTime.now());

            // 处理文档向量（此处可能耗时）
            List<String> vectorIds = processAndStoreDocument(document);
            document.setVectorIds(vectorIds);
            document.setChunkCount(vectorIds.size());

            // 保存 - 乐观锁会自动检查version
            // 如果版本不匹配会抛OptimisticLockingFailureException
            mongoTemplate.save(document);

            log.info("飞书文档同步完成, docToken: {}, title: {}, chunks: {}",
                docToken, title, vectorIds.size());
            return document;

        } catch (org.springframework.dao.OptimisticLockingFailureException e) {
            retry++;
            log.warn("版本冲突，正在重试 {}/{}, docToken: {}", retry, maxRetries, docToken);
            if (retry >= maxRetries) {
                throw new RuntimeException("并发同步失败，请稍后重试", e);
            }
            // 指数退避
            try {
                Thread.sleep(100L * retry);
            } catch (InterruptedException ie) {
                Thread.currentThread().interrupt();
                throw new RuntimeException("同步中断", ie);
            }
        }
    }

    throw new RuntimeException("同步飞书文档失败，超过最大重试次数");
}
```

```java
/**
 * 验证回答与知识库的一致性（幻觉抑制）- 修复置信度计算
 */
public FactCheckResult factCheck(String answer, List<RetrievalResult> retrievalResults) {
    log.info("执行事实核查");

    // 无检索结果 -> 置信度为0（不可信）
    if (retrievalResults == null || retrievalResults.isEmpty()) {
        return FactCheckResult.builder()
            .isFactual(false)
            .confidence(0.0)
            .verifiedSources(0)
            .totalSources(0)
            .build();
    }

    // 统计匹配的知识源
    int matchCount = 0;
    for (RetrievalResult result : retrievalResults) {
        if (answer.contains(result.getTitle()) || containsKeywords(answer, result.getContent())) {
            matchCount++;
        }
    }

    // 置信度计算:
    // - 无匹配: 0.3 (有知识源但回答未引用)
    // - 部分匹配: (matchCount / total) * 0.7 + 0.3 (最高0.7 + 0.3 = 1.0)
    double confidence;
    if (matchCount == 0) {
        confidence = 0.3;  // 有知识库但未引用 -> 置信度降低
    } else {
        confidence = (double) matchCount / retrievalResults.size() * 0.7 + 0.3;
    }

    boolean isFactual = confidence >= 0.6;  // 置信度阈值: 0.6

    log.info("事实核查完成: 匹配度 {}/{}, 置信度: {:.2f}, 可信: {}",
        matchCount, retrievalResults.size(), confidence, isFactual);

    return FactCheckResult.builder()
        .isFactual(isFactual)
        .confidence(confidence)
        .verifiedSources(matchCount)
        .totalSources(retrievalResults.size())
        .build();
}
```

```java
        // 参数验证

        String content = (String) request.get("content");

                "success", false,

                "errorMessage", "文档标题不能为空"
            ));

                "errorMessage", "文档内容不能为空"
            ));

        if (title.length() > 200) {
            return ResponseEntity.badRequest().body(Map.of(
                "success", false,
                "errorCode", "TITLE_TOO_LONG",
                "errorMessage", "标题长度不能超过200字符"
            ));
        }

        // 内容大小限制（10MB）
        if (content.length() > 10 * 1024 * 1024) {
            return ResponseEntity.badRequest().body(Map.of(
                "success", false,
                "errorCode", "CONTENT_TOO_LARGE",
                "errorMessage", "文档内容不能超过10MB"
            ));
        }

        String documentType = (String) request.getOrDefault("documentType", "MANUAL");
        @SuppressWarnings("unchecked")
        List<String> tags = (List<String>) request.getOrDefault("tags", List.of());
        @SuppressWarnings("unchecked")
        Map<String, String> metadata = (Map<String, String>) request.getOrDefault("metadata", Map.of());

        log.info("添加知识文档, title: {}, size: {}", title, content.length());

        try {
            KnowledgeDocument document = knowledgeService.addKnowledgeDocument(
                title, content, documentType, tags, metadata);

            return ResponseEntity.ok(Map.of(
                "success", true,
                "data", document,
                "message", "文档添加成功"
            ));
        } catch (Exception e) {
            log.error("添加文档失败", e);
            return ResponseEntity.status(500).body(Map.of(
                "success", false,
                "errorCode", "ADD_DOCUMENT_ERROR",
                "errorMessage", "文档添加失败: " + e.getMessage()
            ));
        }
    }

    // ... 其他方法保持不变
```

### DocumentOperationHelper.java

```java
package com.ljl.ai.agent.knowledge.helper;

import lombok.extern.slf4j.Slf4j;

import java.util.ArrayList;
import java.util.List;
import java.util.function.Consumer;

/**
 * 文档操作助手 - 提供事务性操作支持
 */
@Slf4j
public class DocumentOperationHelper {

    private List<Runnable> rollbackActions = new ArrayList<>();
    private boolean success = false;

    /**
     * 执行操作并记录回滚点
     */
    public <T> T execute(String operationName,
                        OperationTask<T> task,
                        Consumer<String> rollback) throws Exception {
        try {
            log.info("执行操作: {}", operationName);
            T result = task.execute();
            success = true;
            return result;
        } catch (Exception e) {
            log.error("操作失败: {}, 开始回滚", operationName, e);
            rollback.accept(operationName);
            throw new RuntimeException("操作失败: " + operationName, e);
        }
    }

    /**
     * 记录回滚操作
     */
    public void recordRollback(Runnable action) {
        rollbackActions.add(action);
    }

    /**
     * 执行所有回滚操作
     */
    public void rollbackAll() {
        for (int i = rollbackActions.size() - 1; i >= 0; i--) {
            try {
                rollbackActions.get(i).run();
            } catch (Exception e) {
                log.error("回滚操作失败", e);
            }
        }
    }

    @FunctionalInterface
    public interface OperationTask<T> {
        T execute() throws Exception;
    }
}
```
